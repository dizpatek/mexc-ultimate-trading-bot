leg(int size) =>
    var leg = 0
    newLegHigh = high[size] > ta.highest(size)
    newLegLow = low[size] < ta.lowest(size)
    if newLegHigh
        leg := BEARISH_LEG
    else if newLegLow
        leg := BULLISH_LEG
    leg

startOfNewLeg(int leg) =>
    ta.change(leg) != 0

startOfBearishLeg(int leg) =>
    ta.change(leg) == -1

startOfBullishLeg(int leg) =>
    ta.change(leg) == +1

atan2(y, x) =>
    var float angle = 0.0
    if x > 0
        angle := math.atan(y / x)
        angle
    else if x < 0 and y >= 0
        angle := math.atan(y / x) + math.pi
        angle
    else if x < 0 and y < 0
        angle := math.atan(y / x) - math.pi
        angle
    else if x == 0 and y > 0
        angle := math.pi / 2
        angle
    else if x == 0 and y < 0
        angle := -math.pi / 2
        angle
    angle

min_max_volume(src) =>
    out = (src - ta.lowest(src, 100)) / (ta.highest(src, 100) - ta.lowest(src, 100)) * 100
    math.max(math.min(100, out), 0)
